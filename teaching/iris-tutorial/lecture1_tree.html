Binary Trees

Section heapGS.
  Context `{!heapGS Σ}.

  Inductive tree :=
    | Leaf
    | Node (l : tree) (n : Z) (r : tree).

  Fixpoint isTree (𝑡 : val) (t : tree) : iProp Σ :=
    match t with
    | Leaf =>
        ⌜𝑡 = NONEV⌝
    | Node l n r =>
        ∃ (node : loc) 𝑙 𝑟,
        ⌜𝑡 = SOMEV #node⌝ ∗
        node ↦ (𝑙, #n, 𝑟) ∗
        isTree 𝑙 l ∗
        isTree 𝑟 r
    end.

  Fixpoint tree_inc t :=
    match t with
    | Leaf =>
        Leaf
    | Node l n r =>
        Node (tree_inc l) (n + 1) (tree_inc r)
    end.

  Definition incₜ : val :=
    rec: "inc" "t" :=
      match: "t" with
        NONE =>
          #()
      | SOME "node" =>
          let: "l" := Fst (Fst !"node") in
          let: "n" := Snd (Fst !"node") in
          let: "r" := Snd !"node" in
          "node" <- ("l", "n" + #1, "r") ;;
          "inc" "l" ;;
          "inc" "r"
      end.
