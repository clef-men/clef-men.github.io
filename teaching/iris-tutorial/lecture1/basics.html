Inductive
base_step
    : expr → state → list observation → expr → state → list expr → Prop :=
    RecS : ∀ (f x : binder) (e : expr) (σ : state),
             base_step (rec: f x := e) σ [] (rec: f x := e)%V σ []
  | PairS : ∀ (v1 v2 : val) (σ : state),
              base_step (v1, v2) σ [] (v1, v2)%V σ []
  | InjLS : ∀ (v : val) (σ : state), base_step (InjL v) σ [] (InjLV v) σ []
  | InjRS : ∀ (v : val) (σ : state), base_step (InjR v) σ [] (InjRV v) σ []
  | BetaS : ∀ (f x : binder) (e1 : expr) (v2 : val) (e' : expr) (σ : state),
              e' = subst' x v2 (subst' f (rec: f x := e1) e1)
              → base_step ((rec: f x := e1)%V v2) σ [] e' σ []
  | UnOpS : ∀ (op : un_op) (v v' : val) (σ : state),
              un_op_eval op v = Some v' → base_step (UnOp op v) σ [] v' σ []
  | BinOpS : ∀ (op : bin_op) (v1 v2 v' : val) (σ : state),
               bin_op_eval op v1 v2 = Some v'
               → base_step (BinOp op v1 v2) σ [] v' σ []
  | IfTrueS : ∀ (e1 e2 : expr) (σ : state),
                base_step (if: #true then e1 else e2) σ [] e1 σ []
  | IfFalseS : ∀ (e1 e2 : expr) (σ : state),
                 base_step (if: #false then e1 else e2) σ [] e2 σ []
  | FstS : ∀ (v1 v2 : val) (σ : state),
             base_step (Fst (v1, v2)%V) σ [] v1 σ []
  | SndS : ∀ (v1 v2 : val) (σ : state),
             base_step (Snd (v1, v2)%V) σ [] v2 σ []
  | CaseLS : ∀ (v : val) (e1 e2 : expr) (σ : state),
               base_step (Case (InjLV v) e1 e2) σ [] (e1 v) σ []
  | CaseRS : ∀ (v : val) (e1 e2 : expr) (σ : state),
               base_step (Case (InjRV v) e1 e2) σ [] (e2 v) σ []
  | AllocNS : ∀ (n : Z) (v : val) (σ : state) (l : loc),
                (0 < n)%Z
                → (∀ i : Z, (0 ≤ i)%Z → (i < n)%Z → heap σ !! (l +ₗ i) = None)
                  → base_step (AllocN #n v) σ [] #l 
                      (state_init_heap l n v σ) []
  | FreeS : ∀ (l : loc) (v : val) (σ : state),
              heap σ !! l = Some (Some v)
              → base_step (Free #l) σ [] #() (state_upd_heap <[l:=None]> σ)
                  []
  | LoadS : ∀ (l : loc) (v : val) (σ : state),
              heap σ !! l = Some (Some v) → base_step ! #l σ [] v σ []
  | StoreS : ∀ (l : loc) (v w : val) (σ : state),
               heap σ !! l = Some (Some v)
               → base_step (#l <- w) σ [] #()
                   (state_upd_heap <[l:=Some w]> σ) []
  | XchgS : ∀ (l : loc) (v1 v2 : val) (σ : state),
              heap σ !! l = Some (Some v1)
              → base_step (Xchg #l v2) σ [] v1
                  (state_upd_heap <[l:=Some v2]> σ) []
  | CmpXchgS : ∀ (l : loc) (v1 v2 vl : val) (σ : state) (b : bool),
                 heap σ !! l = Some (Some vl)
                 → vals_compare_safe vl v1
                   → b = bool_decide (vl = v1)
                     → base_step (CmpXchg #l v1 v2) σ [] (
                         vl, #b)%V
                         (if b then state_upd_heap <[l:=Some v2]> σ else σ)
                         []
  | FaaS : ∀ (l : loc) (i1 i2 : Z) (σ : state),
             heap σ !! l = Some (Some #i1)
             → base_step (FAA #l #i2) σ [] #i1
                 (state_upd_heap <[l:=Some #(i1 + i2)]> σ) []
  | ForkS : ∀ (e : expr) (σ : state), base_step (Fork e) σ [] #() σ [e]
  | NewProphS : ∀ (σ : state) (p : proph_id),
                  p ∉ used_proph_id σ
                  → base_step NewProph σ [] #p
                      (state_upd_used_proph_id (union {[p]}) σ) []
  | ResolveS : ∀ (p : proph_id) (v : val) (e : expr) 
                 (σ : state) (w : val) (σ' : state) 
                 (κs : list observation) (ts : list expr),
                 base_step e σ κs v σ' ts
                 → p ∈ used_proph_id σ
                   → base_step (Resolve e #p w) σ 
                       (κs ++ [(p, (v, w))]) v σ' ts.

fix fill_item (Ki : ectx_item) (e : expr) {struct Ki} : expr :=
  match Ki with
  | AppLCtx v2 => e v2
  | AppRCtx e1 => e1 e
  | UnOpCtx op => UnOp op e
  | BinOpLCtx op v2 => BinOp op e v2
  | BinOpRCtx op e1 => BinOp op e1 e
  | IfCtx e1 e2 => (if: e then e1 else e2)%E
  | PairLCtx v2 => (e, v2)%E
  | PairRCtx e1 => (e1, e)%E
  | FstCtx => Fst e
  | SndCtx => Snd e
  | InjLCtx => InjL e
  | InjRCtx => InjR e
  | CaseCtx e1 e2 => Case e e1 e2
  | AllocNLCtx v2 => AllocN e v2
  | AllocNRCtx e1 => AllocN e1 e
  | FreeCtx => Free e
  | LoadCtx => (! e)%E
  | StoreLCtx v2 => (e <- v2)%E
  | StoreRCtx e1 => (e1 <- e)%E
  | XchgLCtx v2 => Xchg e v2
  | XchgRCtx e1 => Xchg e1 e
  | CmpXchgLCtx v1 v2 => CmpXchg e v1 v2
  | CmpXchgMCtx e0 v2 => CmpXchg e0 e v2

Hoare Triples

Hoare triples are actually defined using a more primitive notion: weakest precondition.
On paper, we use the following definition:
{{{ P }}} e {{{ v, RET v; Q }}} ≜ □ (P -∗ WP e {{ Q }})
In Rocq, for technical reasons, we prefer to use a more sophisticated definition, akin to continuation-passing style:
{{{ P }}} e {{{ v, RET v; Q }}} ≜ □ (∀ Φ, P -∗ ▷ (∀ v, Q -∗ Φ v) -∗ WP e {{ Φ }})
Intuitively, WP e {{ Q }} means that: (1) e is safe to execute and (2) if it terminates on value v, Q v holds.
