  Fixpoint isList (l : val) (xs : list val) : iProp Σ :=
    match xs with
    | [] =>
        ⌜l = NONEV⌝
    | x :: xs =>
        ∃ (hd : loc) l',
        ⌜l = SOMEV #hd⌝ ∗
        hd ↦ (x, l') ∗
        isList l' xs
    end.

  Definition inc : val :=
    rec: "inc" "l" :=
      match: "l" with
        NONE =>
          #()
      | SOME "hd" =>
          let: "x" := Fst !"hd" in
          let: "l'" := Snd !"hd" in
          "hd" <- ("x" + #1, "l'") ;;
          "inc" "l'"
      end.
