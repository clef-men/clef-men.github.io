Section dom.
  Context `{FinMapDom K M D}.
  Context {A : Type}.

  Implicit Type m : M A.

  Lemma elem_ofｰdom₁ m k :
    k ∈ dom m →
    is_Some (m !! k).
  Proof.
    rewrite elem_of_dom //.
  Qed.

  Lemma mapｰunionｰdom m1 m2 :
    dom m2 ⊆ dom m1 →
    m1 ∪ m2 = m1.
  Proof.
    intros Hdom.
    apply map_eq => k.
    destruct (m1 !! k) as [x |] eqn:Hm1_lookup.
    - rewrite lookup_union_l' //.
    - rewrite lookup_union_r //.
      rewrite -!not_elem_of_dom in Hm1_lookup |- *.
      set_solver.
  Qed.
End dom.

Section map_Forall.
  Context `{FinMap K M}.
  Context {A : Type}.

  Implicit Type m : M A.

  Lemma map_Forallｰimpl' P1 P2 m :
    map_Forall P1 m →
    ( ∀ k x,
      m !! k = Some x →
      P1 k x →
      P2 k x
    ) →
    map_Forall P2 m.
  Proof.
    rewrite !map_Forall_lookup. auto.
  Qed.

  Lemma map_Forallｰinsert₂' {P m} k x :
    P k x →
    map_Forall P (delete k m) →
    map_Forall P (<[k := x]> m).
  Proof.
    rewrite -insert_delete_eq.
    apply map_Forall_insert_2.
  Qed.

  Lemma map_Forallｰdeleteｰlookup P m k :
    map_Forall P (delete k m) ↔
      ∀ k' x,
      k ≠ k' →
      m !! k' = Some x →
      P k' x.
  Proof.
    rewrite map_Forall_lookup.
    setoid_rewrite lookup_delete_Some.
    naive.
  Qed.
  Lemma map_Forallｰdeleteｰlookup₁ {P m k} k' x :
    map_Forall P (delete k m) →
    k ≠ k' →
    m !! k' = Some x →
    P k' x.
  Proof.
    rewrite map_Forallｰdeleteｰlookup.
    naive.
  Qed.
  Lemma map_Forallｰdeleteｰlookup₂ P m k :
    ( ∀ k' x,
      k ≠ k' →
      m !! k' = Some x →
      P k' x
    ) →
    map_Forall P (delete k m).
  Proof.
    rewrite map_Forallｰdeleteｰlookup //.
  Qed.
End map_Forall.

Section map_Forall2.
  Context `{FinMapDom K M D}.

  Lemma map_Forall2ｰalt {A B R} (m : M A) (𝑚 : M B) :
    map_Forall2 R m 𝑚 ↔
      dom m ≡ dom 𝑚 ∧
        ∀ k x 𝑥,
        m !! k = Some x →
        𝑚 !! k = Some 𝑥 →
        R k x 𝑥.
  Proof.
    split.
    - intros Hm𝑚. split.
      + eapply map_Forall2_dom. done.
      + intros k x 𝑥 Hm_lookup H𝑚_lookup.
        specialize (Hm𝑚 k).
        rewrite Hm_lookup H𝑚_lookup // in Hm𝑚.
        inv Hm𝑚. done.
    - intros (Hdom & Hm𝑚) k.
      specialize (Hm𝑚 k).
      destruct (m !! k) as [x |] eqn:Hm_lookup.
      + destruct (elem_ofｰdom₁ 𝑚 k) as (𝑥 & H𝑚_lookup).
        { apply elem_of_dom_2 in Hm_lookup. set_solver. }
        rewrite H𝑚_lookup. auto.
      + opose proof* (not_elem_of_dom_1 𝑚 k) as H𝑚_lookup.
        { apply not_elem_of_dom in Hm_lookup. set_solver. }
        rewrite H𝑚_lookup //.
  Qed.

  Lemma map_Forall2ｰflip {A B} R (m : M A) (𝑚 : M B) :
    map_Forall2 R m 𝑚 ↔
    map_Forall2 (λ k x 𝑥, R k 𝑥 x) 𝑚 m.
  Proof.
    rewrite !map_Forall2ｰalt. naive.
  Qed.

  Lemma map_Forall2ｰlookupｰNoneｰl {A B R} {m : M A} {𝑚 : M B} k :
    map_Forall2 R m 𝑚 →
    m !! k = None →
    𝑚 !! k = None.
  Proof.
    rewrite -!not_elem_of_dom.
    intros Hdom%map_Forall2_dom.
    set_solver.
  Qed.
  Lemma map_Forall2ｰlookupｰNoneｰr {A B R} {m : M A} {𝑚 : M B} k :
    map_Forall2 R m 𝑚 →
    𝑚 !! k = None →
    m !! k = None.
  Proof.
    rewrite map_Forall2_flip.
    apply map_Forall2ｰlookupｰNoneｰl.
  Qed.

  Lemma map_Forall2ｰlookupｰSome {A B R} {m : M A} {𝑚 : M B} k x 𝑥 :
    map_Forall2 R m 𝑚 →
    m !! k = Some x →
    𝑚 !! k = Some 𝑥 →
    R k x 𝑥.
  Proof.
    rewrite map_Forall2ｰalt. naive.
  Qed.
  Lemma map_Forall2ｰlookupｰSomeｰl {A B R} {m : M A} {𝑚 : M B} k x :
    map_Forall2 R m 𝑚 →
    m !! k = Some x →
      ∃ 𝑥,
      𝑚 !! k = Some 𝑥 ∧
      R k x 𝑥.
  Proof.
    intros Hm𝑚 Hm_lookup.
    apply elem_of_dom_2 in Hm_lookup as Hm_elem.
    apply map_Forall2_dom in Hm𝑚 as Hdom.
    destruct (elem_ofｰdom₁ 𝑚 k) as (𝑥 & H𝑚_lookup); first set_solver.
    exists 𝑥. split; first done.
    eapply map_Forall2ｰlookupｰSome; done.
  Qed.
  Lemma map_Forall2ｰlookupｰSomeｰr {A B R} {m : M A} {𝑚 : M B} k 𝑥 :
    map_Forall2 R m 𝑚 →
    𝑚 !! k = Some 𝑥 →
      ∃ x,
      m !! k = Some x ∧
      R k x 𝑥.
  Proof.
    rewrite map_Forall2_flip.
    apply: map_Forall2ｰlookupｰSomeｰl.
  Qed.

  Lemma map_Forall2ｰinsertｰl {A B R} {m : M A} {𝑚 : M B} k x 𝑥 :
    map_Forall2 R m 𝑚 →
    𝑚 !! k = Some 𝑥 →
    R k x 𝑥 →
    map_Forall2 R (<[k := x]> m) 𝑚.
  Proof.
    intros Hm𝑚 H𝑚_lookup Hx𝑥.
    odestruct map_Forall2ｰlookupｰSomeｰr as (y & Hm_lookup & _); [done.. |].
    rewrite -(insert_id 𝑚 k 𝑥) //.
    apply map_Forall2_insert_2; done.
  Qed.
  Lemma map_Forall2ｰinsertｰr {A B R} {m : M A} {𝑚 : M B} k x 𝑥 :
    map_Forall2 R m 𝑚 →
    m !! k = Some x →
    R k x 𝑥 →
    map_Forall2 R m (<[k := 𝑥]> 𝑚).
  Proof.
    setoid_rewrite map_Forall2_flip.
    apply: map_Forall2ｰinsertｰl.
  Qed.

  Lemma map_Forall2ｰfmapｰl {A B C R} (f : A → C) (m : M A) (𝑚 : M B) :
    map_Forall2 (λ k x 𝑥, R k (f x) 𝑥) m 𝑚 ↔
    map_Forall2 R (f <$> m) 𝑚.
  Proof.
    rewrite !map_Forall2ｰalt dom_fmap.
    setoid_rewrite lookup_fmap_Some.
    naive.
  Qed.
  Lemma map_Forall2ｰfmapｰl₁ {A B C R} (f : A → C) (m : M A) (𝑚 : M B) :
    map_Forall2 (λ k x 𝑥, R k (f x) 𝑥) m 𝑚 →
    map_Forall2 R (f <$> m) 𝑚.
  Proof.
    rewrite map_Forall2ｰfmapｰl //.
  Qed.
  Lemma map_Forall2ｰfmapｰl₂ {A B C R} (f : A → C) (m : M A) (𝑚 : M B) :
    map_Forall2 R (f <$> m) 𝑚 →
    map_Forall2 (λ k x 𝑥, R k (f x) 𝑥) m 𝑚.
  Proof.
    rewrite -map_Forall2ｰfmapｰl //.
  Qed.
  Lemma map_Forall2ｰfmapｰr {A B C R} (f : B → C) (m : M A) (𝑚 : M B) :
    map_Forall2 (λ k x 𝑥, R k x (f 𝑥)) m 𝑚 ↔
    map_Forall2 R m (f <$> 𝑚).
  Proof.
    setoid_rewrite map_Forall2_flip at 1 2.
    apply: map_Forall2ｰfmapｰl.
  Qed.
  Lemma map_Forall2ｰfmapｰr₁ {A B C R} (f : B → C) (m : M A) (𝑚 : M B) :
    map_Forall2 (λ k x 𝑥, R k x (f 𝑥)) m 𝑚 →
    map_Forall2 R m (f <$> 𝑚).
  Proof.
    rewrite map_Forall2ｰfmapｰr //.
  Qed.
  Lemma map_Forall2ｰfmapｰr₂ {A B C R} (f : B → C) (m : M A) (𝑚 : M B) :
    map_Forall2 R m (f <$> 𝑚) →
    map_Forall2 (λ k x 𝑥, R k x (f 𝑥)) m 𝑚.
  Proof.
    rewrite -map_Forall2ｰfmapｰr //.
  Qed.
End map_Forall2.

Section kmap.
  Context `{FinMap K1 M1} `{FinMap K2 M2}.
  Context (f : K1 → K2) `{!Inj (=) (=) f}.

  Abbreviation kmap := (
    kmap (M1 := M1) (M2 := M2)
  ).

  #[local] Lemma NoDupｰfstｰprod_mapｰmap_to_list {A} (m : M1 A) :
    NoDup (prod_map f id <$> map_to_list m).*1.
  Proof.
    rewrite -list_fmap_compose -(list_fmap_ext (f ∘ fst)) // list_fmap_compose.
    apply NoDup_fmap; first done.
    apply NoDup_fst_map_to_list.
  Qed.
  Lemma map_to_listｰkmap {A} (m : M1 A) :
    map_to_list (kmap f m) ≡ₚ prod_map f id <$> map_to_list m.
  Proof.
    apply map_to_list_to_map, NoDupｰfstｰprod_mapｰmap_to_list.
  Qed.
  Lemma kmapｰlist_to_map {A} (l : list (K1 * A)) :
    NoDup l.*1 →
    kmap f (list_to_map l) = list_to_map (prod_map f id <$> l).
  Proof.
    intros Hnodup.
    apply list_to_map_proper.
    { apply NoDupｰfstｰprod_mapｰmap_to_list. }
    rewrite map_to_list_to_map //.
  Qed.
End kmap.

Section map۰oflatten.
  Context `{FinMap K M}.
  Context {A : Type}.

  Definition map۰oflatten (m : M (option A)) :=
    omap id m.

  Lemma lookupｰmap۰oflattenｰNone m k :
    m !! k = None →
    map۰oflatten m !! k = None.
  Proof.
    rewrite lookup_omap => -> //.
  Qed.
  Lemma lookupｰmap۰oflattenｰSomeｰNone m k :
    m !! k = Some None →
    map۰oflatten m !! k = None.
  Proof.
    rewrite lookup_omap => -> //.
  Qed.
  Lemma lookupｰmap۰oflattenｰSomeｰSome {m k} a :
    m !! k = Some (Some a) →
    map۰oflatten m !! k = Some a.
  Proof.
    rewrite lookup_omap_id_Some //.
  Qed.

  Lemma lookupｰmap۰oflattenｰSomeｰinv m k a :
    map۰oflatten m !! k = Some a →
    m !! k = Some (Some a).
  Proof.
    intros Hoflatten_lookup.
    destruct (m !! k) as [[v |] |] eqn:Hm_lookup.
    all: rewrite lookup_omap Hm_lookup /= in Hoflatten_lookup.
    all: congruence.
  Qed.

  Lemma map۰oflattenｰempty m :
    (∀ k o, m !! k = Some o → o = None) →
    map۰oflatten m = ∅.
  Proof.
    intros Hm.
    apply map_empty => k.
    rewrite eq_None_not_Some. intros (a & Hlookup%lookupｰmap۰oflattenｰSomeｰinv).
    naive.
  Qed.

  Lemma map۰oflattenｰunion m1 m2 :
    m1 ##ₘ m2 →
    map۰oflatten (m1 ∪ m2) = map۰oflatten m1 ∪ map۰oflatten m2.
  Proof.
    intros.
    rewrite /map۰oflatten map_omap_union //.
  Qed.

  Lemma map۰oflattenｰinsert {m} k :
    m !! k = None →
    map۰oflatten (<[k := None]> m) = map۰oflatten m.
  Proof.
    intros Hlookup.
    rewrite /map۰oflatten omap_insert_None // delete_id // lookup_omap Hlookup //.
  Qed.
  Lemma map۰oflattenｰupdate {m} k a :
    map۰oflatten (<[k := Some a]> m) = <[k := a]> (map۰oflatten m).
  Proof.
    rewrite /map۰oflatten omap_insert //.
  Qed.

Section map_to_set.

    size (map_to_set f m) ≤ size m.

Definition map_inj `{FinMap K M} {A} (m : M A) :=
  ∀ k1 k2 x,
  m !! k1 = Some x →
  m !! k2 = Some x →
  k1 = k2.

Section map_img.
  Context `{FinMap K M}.
  Context `{FinSet A SA}.

  Implicit Type m : M A.

  Abbreviation map_img := (
    map_img (SA := SA)
  ).

  Lemma mapｰpigeonhole m :
    size (map_img m) < size m →
      ∃ k1 k2 x,
      m !! k1 = Some x ∧
      m !! k2 = Some x ∧
      k1 ≠ k2.
  Proof.
    induction m as [| k x m Hlookup IH] using map_ind => Hsize.
    - rewrite map_size_empty in Hsize. lia.
    - rewrite map_img_insert_notin // map_size_insert_None // in Hsize.
      destruct_decide (map_Exists (λ k' x', x = x') m) as Hm.
      all: rewrite map_Exists_lookup in Hm.
      + destruct Hm as (k' & x_ & Hlookup' & <-).
        exists k, k', x.
        rewrite lookup_insert_eq lookup_insert_ne //. 1: congruence.
        naive.
      + rewrite size_union in Hsize.
        { rewrite disjoint_singleton_l not_elem_of_map_img. naive. }
        rewrite size_singleton in Hsize.
        destruct IH as (k1 & k2 & y & Hlookup_1 & Hlookup_2 & Hk). 1: lia.
        exists k1, k2, y.
        rewrite !lookup_insert_ne //. 1,2: congruence.
  Qed.

  Lemma sizeｰmap_img m :
    size (map_img m) ≤ size m.
  Proof.
    apply sizeｰmap_to_set.
  Qed.
  Lemma sizeｰmap_imgｰinj m :
    map_inj m →
    size (map_img m) = size m.
  Proof.
    intros Hinj.
    pose proof (sizeｰmap_img m) as [(k1 & k2 & x & Hlookup_1 & Hlookup_2 & Hk)%mapｰpigeonhole |]%Nat.le_lteq. 2: done.
    rewrite /map_inj in Hinj. naive.
  Qed.
End map_img.
