    ∃ l,

    l.[prev] ↦ from_option (#ˡ) t (last nodes) ∗

  Lemma xdeque٠pop_backｰspec t nodes :

    }}}

      iSteps.
    - wp۰load.

      rewrite last_snoc /=.
      iDestruct (xdlchainｰsnoc₁ with "Hnodes") as "(Hnodes & Hnode1_prev & Hnode1_next)"; first done.
      wp۰load.
      destruct nodes as [| node2 nodes _] using rev_ind => /=.

      xdlchain (from_option (#ˡ) #l $ last nodes1) nodes2 #l
