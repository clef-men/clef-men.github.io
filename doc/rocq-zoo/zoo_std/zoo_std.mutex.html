  Please Definition mutex۰init t b : iProp Σ :=
    ∃ l γ,
    ⌜t = #l⌝ ∗

    l ↦ᵣ #b ∗

    ∃ b,

    end.
  Please Definition mutex۰inv t P : iProp Σ :=

    ⌜t = #l⌝ ∗

    inv nroot (inv۰inner l γ P).

  Please Definition mutex۰locked t : iProp Σ :=
    ∃ l γ,
    ⌜t = #l⌝ ∗
    l ↪ γ ∗
    locked γ.
