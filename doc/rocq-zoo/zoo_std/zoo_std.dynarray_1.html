  #[local] Definition model' t vs extra : iProp Σ :=
    ∃ l data,
    ⌜t = #l⌝ ∗
    l.[size] ↦ #(length vs) ∗

    " ( %l{} & %data{} & -> & Hl{}_size & Hl{}_data & Hmodel ) ".
