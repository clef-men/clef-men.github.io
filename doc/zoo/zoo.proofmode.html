      fail "not a 'wp'"
  end.

  match pats with
  | [] ⇒
      Some ([IDrop], [])

      '(pat, id) ←
        match pat with
        | IIdent id ⇒
            Some (pat, None)
        | IFresh ⇒
            Some (pat, None)
        | IIntuitionistic (IIdent id as pat) ⇒
            Some (pat, Some id)

            None
        end ;
      '(pats, ids) ← wp۰block۰parse۰aux pats ;
      let pats := pat :: pats in

        if id is Some id then

        else
          ids
      in
      Some (pats, ids)
  end.
