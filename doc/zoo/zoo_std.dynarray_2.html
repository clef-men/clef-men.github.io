  Definition dynarray_2۰model t vs : iProp Σ :=
    ∃ l data elems extra,
    ⌜t = #l⌝ ∗
    l.[size] ↦ #(length vs) ∗

    array۰model data (DfracOwn 1) ((#*@{location} elems) ++ replicate extra §Empty%V) ∗
    [∗ list] elem; v ∈ elems; vs, element۰model elem v.

    " ( %l & %data & %elems & %extra & -> & Hl_size & Hl_data & Hmodel & Helems ) ".
